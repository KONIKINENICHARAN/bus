CREATE TABLE buses (
    bus_id NUMBER PRIMARY KEY,
    from_address VARCHAR2(50),
    to_address VARCHAR2(50),
    journey_date DATE
);

CREATE TABLE bus_schedule (
    schedule_id NUMBER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    bus_id NUMBER,
    from_address VARCHAR2(50),
    to_address VARCHAR2(50),
    journey_date DATE
);

CREATE TABLE passengers (
    passenger_id NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name VARCHAR2(50),
    age NUMBER,
    gender VARCHAR2(10)
);

CREATE TABLE seat_bookings (
    schedule_id NUMBER NOT NULL,
    seat_no NUMBER NOT NULL,
    passenger_id NUMBER,

    CONSTRAINT pk_seat_booking PRIMARY KEY (schedule_id, seat_no)
);
